$paths = [ordered]@{
  'A_Policies-Defender' = 'HKLM:\SOFTWARE\Policies\Microsoft\Windows Defender'
  'B_Policies-WDSC'     = 'HKLM:\SOFTWARE\Policies\Microsoft\Windows Defender Security Center'
  'C_MDM-Defender'      = 'HKLM:\SOFTWARE\Microsoft\PolicyManager\current\device\Defender'
  'D_MDM-WDSC'          = 'HKLM:\SOFTWARE\Microsoft\PolicyManager\current\device\WindowsDefenderSecurityCenter'
}
foreach ($n in $paths.Keys) {
  if (Test-Path $paths[$n]) {
    Write-Host "== $n : CO" -ForegroundColor Yellow
    Get-ChildItem $paths[$n] -Recurse -EA 0 | ForEach-Object {
      $v = Get-ItemProperty $_.PSPath
      $v.PSObject.Properties | Where-Object { $_.Name -notlike 'PS*' } |
        ForEach-Object { "   $($_.Name) = $($_.Value)" }
    }
    (Get-ItemProperty $paths[$n]).PSObject.Properties |
      Where-Object { $_.Name -notlike 'PS*' } | ForEach-Object { "   $($_.Name) = $($_.Value)" }
  } else { Write-Host "-- $n : khong co" -ForegroundColor DarkGray }
}
